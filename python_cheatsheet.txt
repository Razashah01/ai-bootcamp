10. FUNCTIONS

# Basic
def func(param):
    return param

# Default argument
def func(name, age=0):
    return name, age

# *args — collects into tuple
def func(*args):
    return sum(args)

# **kwargs — collects into dict
def func(**kwargs):
    for k, v in kwargs.items():
        print(k, v)

# Lambda — one liner

11. LOGIC FRAMEWORK — HOW TO THINK
----------------------------------------
Every time you see a problem, ask:

  1. What is my INPUT?
  2. What is my OUTPUT?
  3. Do I need to LOOP?
  4. Do I need to FILTER?

Then map to Python tools:

  Do something to every item   → for loop or comprehension
  Filter by condition          → if inside loop/comprehension
  Find highest / lowest        → max() / min()
  Count items                  → len()
  Add up numbers               → sum()
  Sort items                   → sort() or sorted()
  Store key-value pairs        → dict
  Remove duplicates            → set()
  Reuse logic                  → function
  Return multiple values       → return a, b


----------------------------------------
12. COMMON MISTAKES TO AVOID
----------------------------------------
❌  else(condition):       ✅  else:
❌  =<                     ✅  <=
❌  print inside variable  ✅  variable = value, then print
❌  len(list) in loop      ✅  len(item) in loop
❌  sort() returns None    ✅  sorted() returns new list
❌  *args gives list       ✅  *args gives TUPLE
❌  **kwargs gives list    ✅  **kwargs gives DICT

13. FILE HANDLING
----------------------------------------
# Write a file
with open("file.txt", "w") as f:
    f.write("Hello\n")

# Read entire file
with open("file.txt", "r") as f:
    content = f.read()

# Read line by line
with open("file.txt", "r") as f:
    for line in f:
        print(line.strip())

# Read all lines as list
with open("file.txt", "r") as f:
    lines = f.readlines()

# Append to file
with open("file.txt", "a") as f:
    f.write("New line\n")

# Modes:
# "r" → read
# "w" → write (overwrites existing)
# "a" → append (keeps existing)

15. OOP — CLASSES

# Basic class
class MyClass:
    def __init__(self, name, value):
        self.name  = name      # attribute
        self.value = value     # attribute

    def describe(self):        # method
        print(f"{self.name}: {self.value}")

# Create object
obj = MyClass("Test", 42)
obj.describe()

# Inheritance
class Parent:
    def __init__(self, name):
        self.name = name

    def speak(self):
        print(f"{self.name} speaks")

class Child(Parent):
    def __init__(self, name, age):
        super().__init__(name)   # call parent __init__
        self.age = age

    def speak(self):             # override parent method
        print(f"{self.name} is {self.age}")

# Key rules:
# self        → refers to the object itself
# __init__    → runs automatically when object created
# super()     → calls parent class method
# All methods need self as first parameter

import pandas as pd

# Create DataFrame
df = pd.DataFrame({
    "name":     ["BERT", "GPT", "LLaMA"],
    "accuracy": [0.94, 0.98, 0.91],
    "params":   [110, 345, 700]
})

# DataFrame info
df.shape          # (rows, cols)
df.columns        # column names
df.dtypes         # data types
df.head(3)        # first 3 rows
df.tail(3)        # last 3 rows
df.info()         # summary
df.describe()     # statistics

# Accessing data
df["name"]                    # single column
df[["name", "accuracy"]]      # multiple columns
df.iloc[0]                    # first row
df.iloc[-1]                   # last row
df.iloc[0:3]                  # first 3 rows

# Filtering
df[df["accuracy"] > 0.93]
df[df["task"] == "Generation"]
df[df["year"] == 2023]

# Adding columns
df["new_col"] = df["accuracy"] * 100
df["size"] = np.where(df["params"] > 300, "Large", "Small")

# Dropping
df.drop("col_name", axis=1)   # drop column
df.drop(0, axis=0)            # drop row

# Sorting
df.sort_values("accuracy", ascending=False)
df.sort_values(["year", "accuracy"], ascending=[True, False])

# GroupBy
df.groupby("task")["accuracy"].mean()
df.groupby("year")["model"].apply(list)

# Missing values
df.isnull().sum()                          # count missing
df.dropna()                                # drop rows with missing
df["col"].fillna(df["col"].mean())         # fill with mean
df["col"].fillna("Unknown")               # fill with value

# CSV
df.to_csv("file.csv", index=False)        # save
df = pd.read_csv("file.csv")             # load

# Finding max/min row
df.loc[df["accuracy"].idxmax(), "model"]  # name of highest
df.loc[df["accuracy"].idxmin(), "model"]  # name of lowest

# Iterate rows
for _, row in df.iterrows():
    print(row["name"], row["accuracy"])
